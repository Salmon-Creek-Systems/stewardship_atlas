server {
	root /root/swales_dev/kennedy/;

	index index.html index.htm index.nginx-debian.html;

	server_name kennedy.fireatlas.org;

	location ~ "/upload/([0-9a-zA-Z-.]*)$" {
        dav_methods  PUT DELETE MKCOL COPY MOVE;
        client_body_temp_path /tmp/incoming;
        alias     /data/files/incoming/$1;
        create_full_put_path   on;
        dav_access             group:rw  all:r;

        client_body_in_file_only on;
        client_body_buffer_size 128k;
        client_max_body_size 100M;
    }

        location = / {
	return 302 /staging/outlets/html/public;
}
	location / {
		try_files $uri $uri/ =404;
	}

	location /staging/outlets/html/admin/ {
		auth_basic "VFD Admin Login";
        	auth_basic_user_file /root/swales/roles/kennedy/admin.htpasswd;
	}

	location /staging/outlets/html/internal/ {
		auth_basic "VFD Admin Login";
        	auth_basic_user_file /root/swales/roles/kennedy/internal.htpasswd;
	}

    listen 443 ssl; # managed by Certbot
    ssl_certificate /etc/letsencrypt/live/kennedy.fireatlas.org/fullchain.pem; # managed by Certbot
    ssl_certificate_key /etc/letsencrypt/live/kennedy.fireatlas.org/privkey.pem; # managed by Certbot
    include /etc/letsencrypt/options-ssl-nginx.conf; # managed by Certbot
    ssl_dhparam /etc/letsencrypt/ssl-dhparams.pem; # managed by Certbot
}
